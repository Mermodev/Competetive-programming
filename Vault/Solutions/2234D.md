#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    long long N, K;
    cin >> N >> K;
    string a, b;
    cin >> a >> b;
    long long Zeros[3] = {0, 0, 0}, Ones[3] = {0, 0, 0};
    for(int i = 0; i < N; i++){ 
      (a[i] == '0') ? Zeros[0]++ : Ones[0]++;
      (b[i] == '0') ? Zeros[1]++ : Ones[1]++;
      (a[i] == b[i]) ? Zeros[2]++ : Ones[2]++;
    }
    long long Score[3];
    for(int i = 0; i < 3; i++)
      Score[i] = Ones[i] * Zeros[i];
    if(K % 2 == 0)
      cout << (Score[0] + Score[1] + Score[2]) * (((1 << K) + 1) / 3) + (Score[0] + Score[1]) << '\n';
    else 
      cout << (Score[0] + Score[1] + Score[2]) * (((1 << K) + 1) / 3) << '\n';
  }
}
