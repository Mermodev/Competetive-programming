#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N, K;
    cin >> N >> K;
    string S;
    cin >> S;
    vector<int> Score = {0, 0};
    for(int i = 0; i < 2 * N; i++){
      if(S[i] == '0')
        continue;
      if(S[(i + 1) % (2 * N)] == '1')
        Score[(i % 2 == 0) ? 1 : 0]++;
      else 
        Score[i % 2]++;
    }
    cout << Score[0] << ' ' << Score[1] << '\n';
  }
}
