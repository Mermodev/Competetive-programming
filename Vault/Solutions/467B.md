```cpp
#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int N, M, K;
  cin >> N >> M >> K;
  vector<long long> Players(M + 1);
  for(auto&e : Players)
    cin >> e;
  int Ans = 0;
  for(int i = 0; i < M; i++){
    auto e = Players[i];
    long long copy = Players[M];
    int diff = 0;
    while(diff <= K && (e || copy)){
      if((e & 1) != (copy & 1))
        diff++;
      e >>= 1;
      copy >>= 1;
    }
    if(diff <= K)
      Ans++;
  }
  cout << Ans;
}
```
