#include<bits/stdc++.h>
using namespace std;

long long MOD = 1e9 + 7;

void Calc_dp(vector<vector<int>>&G, vector<long long>&dp, int C){
  for(auto&e : G[C]){
    Calc_dp(G, dp, e);
    dp[C] = (dp[C] + dp[e] + 1) % MOD;
  }
  dp[C] = (dp[C] + 1) % MOD;
}

void Calc_ans(vector<vector<int>>&G, vector<long long>&dp, vector<long long>&ans, int C, int P){
  ans[C] = (dp[C] + ans[P]) % MOD;
  for(auto&e : G[C])
    Calc_ans(G, dp, ans, e, C);
}

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N;
    cin >> N;
    vector<vector<int>> G(N + 1);
    for(int i = 1; i <= N; i++){
      int l, r;
      cin >> l >> r;
      if(l == 0)
        continue;
      G[i].emplace_back(l);
      G[i].emplace_back(r);
    }
    vector<long long> dp(N + 1), ans(N + 1);
    Calc_dp(G, dp, 1);
    Calc_ans(G, dp, ans, 1, 0);
    for(int i = 1; i < N + 1; i++)
      cout << ans[i] << ' ';
    cout << '\n';
  }
}
