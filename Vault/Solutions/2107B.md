#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N, K;
    cin >> N >> K;
    vector<int> V(N);
    for(auto&e : V)
      cin >> e;
    sort(V.begin(), V.end());
    V[V.size() - 1]--;
    sort(V.begin(), V.end());
    long long Sum = 1;
    for(auto&e : V)
      Sum += e;
    if(V[V.size() - 1] - V[0] > K || Sum % 2 == 0)
      cout << "Jerry\n";
    else 
      cout << "Tom\n";
  }
}
