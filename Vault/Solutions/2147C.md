#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N;
    cin >> N;
    string S;
    cin >> S;
    bool ok = true;
    bool curr = (S[0] == '1');
    int cnt = 0;
    for (int i = 0; i < N; i++) {
      if (S[i] == '0')
        cnt++;
      if (i == 0)
        continue;
      if (S[i] == S[i - 1] && S[i] == '0')
        curr = false;
      if (S[i] == S[i - 1] && S[i] == '1') {
        if (curr && cnt % 2 == 1)
          ok = false;
        curr = true;
        cnt = 0;
      }
    }
    if (curr && cnt % 2 == 1 && S[N - 1] == '1')
      ok = false;
    cout << (ok ? "YES" : "NO") << "\n";
  }

}
