#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N, D;
    cin >> N >> D;
    vector<long long> Table(N * 3);
    for(int i = 0; i < N; i++){
      cin >> Table[i];
      Table[i + N] = Table[i];
      Table[i + 2 * N] = Table[i];
    }
    long long LeftSum = 0, RightSum = 0;
    long long Ans = 0;
    for(int k = 1; k <= D; k++){
      LeftSum += Table[N - k];
      RightSum += Table[N + k];
    }
    for(int i = N; i < 2 * N; i++){
      if(2 * D * Table[i] > LeftSum + RightSum)
        Ans += 2 * D * Table[i] - LeftSum - RightSum;
      RightSum += Table[i + D + 1] - Table[i + 1];
      LeftSum += Table[i] - Table[i - D];
    }
    cout << Ans << '\n';
  }
}
