#include<bits/stdc++.h>
using namespace std;

struct Node{
  int bit, size;
  vector<long long> children;
  Node(int bit){
    this->bit = bit;
    size = 0;
    children = {-1, -1};
  }
};

const int DEPTH = 20;

vector<Node> Trie;

void Add(int value){
  int Curr = 0;
  for(int i = 0; i < DEPTH; i++){
    int b = (value >> i) & 1;
    if(Trie[Curr].children[b] == -1){
      Trie[Curr].children[b] = Trie.size();
      Trie.push_back(Node(b));
      Trie[Curr].size++;
    }
    Curr = Trie[Curr].children[b];
  }
}

int Find(int value){
  int Curr = 0, Ans = 0;
  vector<int> path(DEPTH), chosenBit(DEPTH);
  for(int i = 0; i < DEPTH; i++){
    path[i] = Curr;
    int want = 1 - ((value >> i) & 1);
    int b = (Trie[Curr].children[want] != -1) ? want : 1 - want;
    chosenBit[i] = b;
    Ans |= (b << i);
    Curr = Trie[Curr].children[b];
  }
  for(int i = DEPTH - 1; i >= 0; i--){
    int node = path[i], b = chosenBit[i];
    int child = Trie[node].children[b];
    if(child == -1 || Trie[child].size == 0){
      Trie[node].size--;
      Trie[node].children[b] = -1;
    }
  }
  return Ans;
}

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    Trie.clear();
    Trie.push_back(Node(-1));
    int L, R;
    cin >> L >> R;
    for(int i = L; i <= R; i++)
      Add(i);
    long long Ans = 0;
    vector<long long> Assign(R - L + 1);
    for(long long i = R; i >= L; i--){
      long long temp = Find(i);
      Ans += i | temp;
      Assign[i - L] = temp;
    } 
    cout << Ans << '\n';
    for(auto&e : Assign)
      cout << e << ' ';
    cout << '\n';
  }
}
