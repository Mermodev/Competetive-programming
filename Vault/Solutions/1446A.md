#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    long long N, W;
    cin >> N >> W;
    vector<pair<long long, long long>> V(N);
    for(int i = 0; i < N; i++){
      cin >> V[i].first;
      V[i].second = i + 1;
    }
    sort(V.begin(), V.end());
    vector<int> Ans;
    long long Sum = 0;
    while(Sum < (W + 1) / 2 && V.size() != 0){
      if(Sum + V[V.size() - 1].first <= W){
        Sum += V[V.size() - 1].first;
        Ans.emplace_back(V[V.size() - 1].second);
      }
      V.pop_back();
    }
    if(Sum < (W + 1) / 2){
      cout << "-1\n";
      continue;
    }
    cout << Ans.size() << '\n';
    for(int i = Ans.size() - 1; i >= 0; i--)
      cout << Ans[i] << ' ';
    cout << '\n';
  }
}
