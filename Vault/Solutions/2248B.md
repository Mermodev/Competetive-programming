#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N, M;
    bool Possible = true;
    cin >> N >> M;
    vector<int> A(N), B(M);
    for(auto&e : A)
      cin >> e;
    for(auto&e : B)
      cin >> e;
    sort(A.begin(), A.end());
    sort(B.begin(), B.end());
    vector<int> newA;
    if(N <= M){
      cout << "NO\n";
      continue;
    }
    for(int i = 0; i < M; i++)
      if(B[i] < A[i]){
        Possible = false;
        break;
      }
    if(!Possible){
      cout << "NO\n";
      continue;
    } 
    for(int i = M; i < N; i++)
      newA.emplace_back(A[i]);
    int Pointer = 0;
    for(int i = 0; i < M; i++){
      while(Pointer < newA.size() && newA[Pointer] < B[i])
        Pointer++;
      if(Pointer == newA.size()){
        Possible = false;
        break;
      }
      Pointer++;
    }
    if(Possible)
      cout << "YES\n";
    else 
      cout << "NO\n";
  }
}
