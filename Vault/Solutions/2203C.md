#include<bits/stdc++.h>
using namespace std;

bool Check(long long S, long long M, long long&Mid){
  long long Accumulated = 0;
  for(int i = 60; i >= 0; i--){
    Accumulated <<= 1;
    if((S >> i) & 1) Accumulated++;
    if((M >> i) & 1) Accumulated -= min(Accumulated, Mid);
  }
  return Accumulated == 0;
}

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    long long S, M;
    cin >> S >> M;
    long long Left = 0;
    long long Right = 1ll << 60;
    long long Mid = (Left + Right) >> 1;
    if(!Check(S, M, Right)){
      cout << -1 << '\n';
      continue;
    }
    while(Left != Right){
      if(Check(S, M, Mid))
        Right = Mid;
      else 
        Left = Mid + 1;
      Mid = (Left + Right) >> 1;
    }
    cout << Mid << '\n';
  }
}
