#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N;
    cin >> N;
    string S;
    cin >> S;
    int Ans = 1;
    for(int i = 1; i < N; i++)
      if(S[i] != S[i - 1])
        Ans++;
    bool Done = false;
    for(int i = 1; i < N - 1; i++)
      if(S[i] != S[i - 1] && S[i] != S[i + 1] && S[i - 1] == S[i + 1]){
        Ans -= 2;
        Done = true;
        break;
      }
    for(int i = 1; i < N - 1 && !Done; i++)
      if(S[i] != S[i - 1] && S[i] != S[i + 1]){
        Ans--;
        Done = true;
        break;
      }
    cout << Ans << '\n';
  }
}
