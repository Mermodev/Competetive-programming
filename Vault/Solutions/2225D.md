#include<bits/stdc++.h>
using namespace std;

long long MOD = 998244353;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    long long N, X;
    cin >> N >> X;
    long long ZLeft = X / 4 + 1, ZRight = (N + 1) / 4 - ZLeft + 1;
    long long OLeft = (X + 2) / 4, ORight = (N + 3) / 4 - OLeft;
    long long Ans = ((ZLeft % MOD) * (ZRight % MOD)) % MOD + ((OLeft % MOD) * (ORight % MOD)) % MOD;
    cout << Ans % MOD << '\n';
  }
}
