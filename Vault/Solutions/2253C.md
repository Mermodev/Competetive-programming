#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N, M, X, Y;
    cin >> N >> M >> X >> Y;
    vector<int> A(X), B(Y);
    map<int, int> Assigment;
    for(auto&e : A){
      cin >> e;
      Assigment[e] = 1;
    }
    for(auto&e : B){
      cin >> e;
      if(Assigment[e] == 0)
        Assigment[e] = 2;
      if(Assigment[e] == 1)
        Assigment[e] = 3;
    }
    vector<pair<int, int>> Choosable;
    for(auto&element : Assigment)
      Choosable.emplace_back(element);
    sort(Choosable.begin(), Choosable.end());
    int rem_rows = min(N, X);
    int rem_cols = min(M, Y);
    int total_rem = rem_cols + rem_rows;
    if(total_rem == N + M)
      total_rem--;
    long long Sum = 0;
    for(int i = Choosable.size() - 1; i >= 0 && total_rem > 0; i--){
      if(Choosable[i].second == 1 && rem_rows > 0){
        rem_rows--;
        Sum += Choosable[i].first;
        total_rem--;
      }
      if(Choosable[i].second == 2 && rem_cols > 0){
        rem_cols--;
        Sum += Choosable[i].first;
        total_rem--;
      }
      if(Choosable[i].second == 3){
        total_rem--;
        Sum += Choosable[i].first;
      }
    }
    cout << Sum << '\n';
  }
}
