#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N;
    cin >> N;
    multiset<long long> Ms;
    long long Total = 0;
    for(int i = 0; i < N; i++){
      long long X;
      cin >> X;
      Ms.insert(X);
      Total += X;
    }
    if(Total <= 0){
      cout << -1 << '\n';
      continue;
    }
    vector<long long> Ans(N);
    long long S = 0;
    bool Ok = true;
    for(int i = 0; i < N; i++){
      auto It = Ms.upper_bound(-S);
      if(It == Ms.end()){
        Ok = false;
        break;
      }
      S += *It;
      Ms.erase(It);
      Ans[i] = S;
    }
    if(!Ok){
      cout << -1 << '\n';
      continue;
    }
    for(int i = 0; i < N; i++)
      cout << Ans[i] << " \n"[i == N - 1];
  }
}
