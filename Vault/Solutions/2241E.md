#include<bits/stdc++.h>
using namespace std;

void dfs(int C, vector<vector<int>>&G, vector<int>&SubTree){
  SubTree[C] = 1;
  for(auto&e : G[C]){
    if(SubTree[e] != 0)
      continue;
    dfs(e, G, SubTree);
    SubTree[C] += SubTree[e];
  }
}

long long Calc_score(vector<int>&Base){
  vector<long long> Take_one(Base.size()), Take_two(Base.size()), Take_three(Base.size());
  for(int i = 0; i < Base.size(); i++){
    Take_one[i] = Base[i];
    if(i > 0)
      Take_one[i] += Take_one[i - 1];
  }
  for(int i = 1; i < Base.size(); i++)
    Take_two[i] = Take_two[i - 1] + Take_one[i - 1] * Base[i];
  for(int i = 2; i < Base.size(); i++)
    Take_three[i] = Take_three[i - 1] + Take_two[i - 1] * Base[i];
  return Take_two[Base.size() - 1] + Take_three[Base.size() - 1];
}

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N;
    cin >> N;
    vector<int> Vals(N), SubTree(N);
    for(auto&e : Vals)
      cin >> e;
    vector<vector<int>> G(N);
    for(int i = 1; i < N; i++){
      int u, v;
      cin >> u >> v;
      G[--u].emplace_back(--v);
      G[v].emplace_back(u);
    }
    dfs(0, G, SubTree);
    long long Ans = 0;
    for(int i = 0; i < N; i++){
      if(pow((int)sqrt(Vals[i]), 2) != Vals[i])
        continue;
      vector<int> Base;
      for(auto&e : G[i]){
        if(SubTree[e] > SubTree[i])
          Base.emplace_back(N - SubTree[i]);
        else
          Base.emplace_back(SubTree[e]);
      }
      Ans += Calc_score(Base);
    }
    cout << Ans << '\n';
  }
}
