#include<bits/stdc++.h>
using namespace std;

int main(){
  int Q;
  cin >> Q;
  while(Q--){
    int N, M;
    cin >> N;
    M = N;
    long long Sum = 0;
    vector<pair<int, int>> Divisors;
    for(int i = 2; i < sqrt(M) + 3; i++){
      if(N % i == 0)
        Divisors.emplace_back(make_pair(i, 0));
      while(N % i == 0){
        Divisors[Divisors.size() - 1].second++;
        N /= i;
        Sum++;
      }
    }
    if(N > 1){
      Sum++;
      Divisors.emplace_back(make_pair(N, 1));
    }
    cout << Sum + Divisors.size() - 1 << '\n';
  }
}
