#include<bits/stdc++.h>
using namespace std;

pair<long long, long long> Reverse(pair<long long, long long> Original){
  return make_pair(Original.second, Original.first);
}

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int Robot_amt, Spike_amt, Command_amt, Temp;
    cin >> Robot_amt >> Spike_amt >> Command_amt;
    set<long long> Robot_pos, Spike_pos, Inv_Spike_pos;
    for(int i = 0; i < Robot_amt; i++){
      cin >> Temp;
      Robot_pos.insert(Temp);
    }
    for(int i = 0; i < Spike_amt; i++){
      cin >> Temp;
      Spike_pos.insert(Temp);
      Inv_Spike_pos.insert(-Temp);
    }
    multiset<pair<long long, long long>> Bound_LR, Bound_RL;
    for(auto& Robot : Robot_pos){
      long long Left = (Inv_Spike_pos.lower_bound(-Robot) == Inv_Spike_pos.end()) ? -INT_MAX : -*Inv_Spike_pos.lower_bound(-Robot);
      long long Right = (Spike_pos.lower_bound(Robot) == Spike_pos.end()) ? INT_MAX : *Spike_pos.lower_bound(Robot);
      Bound_LR.insert(make_pair(Robot - Left, Right - Robot));
      Bound_RL.insert(make_pair(Right - Robot, Robot - Left));
    }
    int Delta = 0;
    string Command;
    cin >> Command;
    for(int i = 0; i < Command_amt; i++){
      Delta += (Command[i] == 'L') ? -1 : 1;
      while(Bound_LR.size() && (*(Bound_LR.begin())).first <= -Delta){
        Bound_RL.erase(Bound_RL.find(Reverse(*(Bound_LR.begin()))));
        Bound_LR.erase(Bound_LR.begin());
      }
      while(Bound_RL.size() && (*(Bound_RL.begin())).first <= Delta){
        Bound_LR.erase(Bound_LR.find(Reverse(*(Bound_RL.begin()))));
        Bound_RL.erase(Bound_RL.begin());
      }
      cout << Bound_LR.size() << ' ';
    }
    cout << '\n';
  }
}
