```cpp
#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int T;
  cin >> T;
  while(T--){
    int N, Q;
    cin >> N >> Q;
    vector<int> A(N);
    vector<pair<int,int>> B(N);
    for(int i = 0; i < N; i++){
      cin >> A[i];
      B[i] = {A[i], i};
    }
    sort(B.begin(), B.end());
    long long Mx = 0;
    for(int i = 0; i < N; i++)
      Mx = max(Mx, (long long)(i ^ B[i].second));
    if(Mx == 0){
      cout << 0 << '\n';
      continue;
    }
    cout << (1LL << (63 - __builtin_clzll(Mx))) << '\n';
  }
}
```
