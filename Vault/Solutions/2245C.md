#include<bits/stdc++.h>
using namespace std;

void Solve(){
  int N, M, K;
  cin >> N >> K;
  M = N;
  long long Curr = 1;
  set<int> Pushed;
  Pushed.insert(0);
  Pushed.insert(N);
  while(Curr < N){
    if(M % 2 != K % 2){
      if(Curr == N){
        cout << "NO\n";
        return;
      }
      Pushed.insert(Curr);
    }
    Curr <<= 1;
    M >>= 1;
    K >>= 1;
  }
  if(K != M){
    cout << "NO\n";
    return;
  }
  cout << "YES\n";
  for(int i = 0; i < N; i++)
    if(Pushed.find(i) == Pushed.end())
      cout << i << ' ';
  for(auto&e : Pushed)
    if(e != N)
      cout << e << ' ';
  cout << '\n';
}

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    Solve();
  }
}
