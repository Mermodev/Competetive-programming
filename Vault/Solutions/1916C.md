#include<bits/stdc++.h>
using namespace std;

int main(){
  cin.tie(0) -> sync_with_stdio(0);
  int Q;
  cin >> Q;
  while(Q--){
    int N, nOdd = 0, nEven = 0;
    long long Sum = 0, temp;
    cin >> N >> temp;
    if(temp & 1) nOdd++;
    else         nEven++;
    Sum = temp;
    cout << Sum << ' ';
    for(int i = 1; i < N; i++){
      cin >> temp;
      if(temp & 1) nOdd++;
      else         nEven++;
      Sum += temp;
      if(nEven == 0)
        cout << Sum - ((nOdd - 2) / 3 + (nOdd - 2) % 3) << ' ';
      else 
        cout << Sum - (nOdd / 3 + (nOdd % 3 == 1)) << ' ';
    }
    cout << '\n';
  }
}
